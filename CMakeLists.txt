cmake_minimum_required(VERSION 3.18)
project(gpuforge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(KAMA_MEMORY_POOL_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(KAMA_MEMORY_POOL_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(KAMA_MEMORY_POOL_ENABLE_STATS ON CACHE BOOL "" FORCE)
add_subdirectory(vendor_memorypool/v3 vendor_memorypool-build EXCLUDE_FROM_ALL)
option(GPUFORGE_ENABLE_CUDA "Build CUDA kernels" ON)
option(GPUFORGE_BUILD_TESTS "Build tests" ON)
add_library(gpuforge STATIC src/tensor.cpp src/kernels.cpp src/runtime.cpp src/kv_cache.cpp src/graph.cpp src/compiler.cpp src/memory.cpp src/telemetry.cpp src/tuning.cpp src/pipeline.cpp src/precision.cpp src/shape.cpp src/planner.cpp src/executor.cpp src/collective.cpp src/scheduler.cpp src/benchmark.cpp src/autotune.cpp)
target_include_directories(gpuforge PUBLIC include)
target_link_libraries(gpuforge PUBLIC KamaMemoryPool::memory_pool)
target_compile_options(gpuforge PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-O3 -march=native>)
if(GPUFORGE_ENABLE_CUDA)
  enable_language(CUDA OPTIONAL)
  if(CMAKE_CUDA_COMPILER)
    target_sources(gpuforge PRIVATE cuda/kernels.cu)
    target_compile_definitions(gpuforge PUBLIC GPUFORGE_CUDA=1)
    set_target_properties(gpuforge PROPERTIES CUDA_STANDARD 17 CUDA_SEPARABLE_COMPILATION ON CUDA_RESOLVE_DEVICE_SYMBOLS ON)
  endif()
endif()
add_executable(gpuforge-bench examples/bench.cpp)
target_link_libraries(gpuforge-bench PRIVATE gpuforge)
if(GPUFORGE_BUILD_TESTS)
  enable_testing()
  add_executable(gpuforge-test tests/test_core.cpp)
  target_link_libraries(gpuforge-test PRIVATE gpuforge)
  add_test(NAME core COMMAND gpuforge-test)
endif()
